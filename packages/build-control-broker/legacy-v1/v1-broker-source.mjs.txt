import { createHash, randomUUID } from 'node:crypto';
import { spawn as nodeSpawn } from 'node:child_process';
import { lstat, mkdir, open, readFile, readdir, realpath, rename, rm, stat, statfs, unlink } from 'node:fs/promises';
import net from 'node:net';
import { dirname, isAbsolute, join, normalize, relative } from 'node:path';
import { BUILD_CONTROL_PROTOCOL, SOURCE_RUNTIME_IDENTITY, externalRoots, immutableToolRoot, normalizeManagedAgentRequest, normalizeRuntimeIdentity, validateManagedRustEnvironment, validateManagedRustupInvocation } from '@rightkit/build-control-core';

import { reapPersistedProcessTree, spawnManagedProcess, startManagedProcess, terminateProcessTree } from './process-tree.mjs';

const byteCount = (value, field) => { if (!Number.isSafeInteger(value) || value < 0) throw new TypeError(`${field} must be a non-negative safe integer`); return value; };
const contained = (root, value) => { const diff = relative(root, value); return diff === '' || (!diff.startsWith('..') && !isAbsolute(diff)); };
const digest = (value) => createHash('sha256').update(value).digest('hex');
const dimension = (value, field) => { if (typeof value !== 'string' || !value) throw new TypeError(`${field} is required`); return value; };

/** State & execution runtime. Storage identity has already been attested by caller. */
export class BuildControlBroker {
  #root; #capacity; #used; #reservations = new Map(); #leases = new Map(); #entries = new Map(); #stateFile; #receiptsDir; #ready; #writes = Promise.resolve(); #reconciliation; #gcRun = null;
  constructor({ externalRoot, volumeId, capacityBytes, usedBytes = 0, stateFile, receiptsDir }) {
    if (!externalRoot || typeof externalRoot !== 'object') throw new TypeError('managed external root is required');
    if (!isAbsolute(externalRoot.path) || !externalRoot.volumeId || externalRoot.mounted !== true || externalRoot.writable !== true || externalRoot.internal === true) throw new TypeError('managed external root must be mounted, writable, external & volume-identified');
    this.#root = Object.freeze({ path: normalize(externalRoot.path), volumeId: String(externalRoot.volumeId) });
    if (volumeId !== undefined && volumeId !== this.#root.volumeId) throw new TypeError('external root volume identity mismatch');
    this.#capacity = byteCount(capacityBytes, 'capacityBytes'); this.#used = byteCount(usedBytes, 'usedBytes');
    if (this.#used > this.#capacity) throw new RangeError('used bytes exceed capacity');
    if (!stateFile || !receiptsDir) throw new TypeError('durable state file & receipts directory are required');
    this.#stateFile = normalize(stateFile); this.#receiptsDir = normalize(receiptsDir);
    if (!contained(this.#root.path, this.#stateFile)) throw new TypeError('state file must remain under external root');
    if (!contained(this.#root.path, this.#receiptsDir)) throw new TypeError('receipts directory must remain under external root');
    this.#ready = this.#load();
  }
  async ready() { await this.#ready; }
  async waitForMaintenance() { await this.#ready; while (this.#gcRun) await this.#gcRun; }
  async enqueue(input) { await this.waitForMaintenance(); return this.queue(input); }
  async flush() { await this.#ready; await this.#writes; }
  entry(id) { return this.#entries.get(id) ?? null; }
  get hasOpenRequests() { return [...this.#entries.values()].some((entry) => ['QUEUED', 'ACTIVE'].includes(entry.state)); }
  get capacity() { return Object.freeze({ capacityBytes: this.#capacity, usedBytes: this.#used, reservedBytes: this.#reservedBytes(), availableBytes: this.#availableBytes(), volumeId: this.#root.volumeId }); }
  rootsFor({ repositoryIdentity, toolchain, targetTriple, buildProfile }) {
    const repository = dimension(repositoryIdentity, 'repositoryIdentity'); const tool = dimension(toolchain, 'toolchain'); const target = dimension(targetTriple, 'targetTriple'); const profile = dimension(buildProfile, 'buildProfile');
    // Cargo metadata must report one target directory for every profile. Cargo
    // keeps debug/release outputs below that root; partitioning the root by
    // profile makes Tauri compile into one tree then lipo from another.
    const cargoStorageProfile = 'release';
    const base = externalRoots({ externalRoot: this.#root.path, repositoryIdentity: repository, toolchain: tool, target, buildProfile: cargoStorageProfile, policyProfile: 'MANAGED_AGENT' });
    const toolKey = digest(tool).slice(0, 24); const profileKey = digest(cargoStorageProfile).slice(0, 24);
    const roots = { ...base, stage: join(base.build, 'stage'), target: join(base.targets, target, profileKey), cargoHome: base.cargoHome, packageCache: join(base.root, 'package-cache'), rustupHome: join(this.#root.path, 'rightkit-build-control', 'global', 'toolchains', toolKey), sccache: base.compilerCache, sccacheSocket: join(this.#root.path, '.rk-sccache.sock') };
    const pathValues = ['root', 'cargoHome', 'compilerCache', 'targets', 'build', 'temp', 'stage', 'target', 'packageCache', 'rustupHome', 'sccache', 'sccacheSocket'];
    if (!pathValues.every((key) => contained(this.#root.path, roots[key]))) throw new Error('computed root escapes managed external root');
    return Object.freeze(roots);
  }
  queue({ request, repositoryIdentity, toolchain, targetTriple, buildProfile, reservationBytes, target, writer = true, id = randomUUID() }) {
    const normalized = normalizeManagedAgentRequest(request); const reserve = byteCount(reservationBytes, 'reservationBytes'); const roots = this.rootsFor({ repositoryIdentity, toolchain, targetTriple, buildProfile });
    if (!id || this.#entries.has(id)) throw new TypeError('unique request id is required');
    const targetName = dimension(target ?? targetTriple, 'target'); if (reserve > this.#capacity) throw new RangeError('reservation exceeds hard quota');
    const entry = Object.freeze({ id, request: normalized, repositoryIdentity, toolchain, targetTriple, buildProfile, target: targetName, writer: Boolean(writer), reservationBytes: reserve, leaseKey: `${roots.namespace}:${targetName}:cargo`, roots, state: 'QUEUED', queuedAt: new Date().toISOString() });
    this.#entries.set(id, entry); this.#queuePersist(); return entry;
  }
  activate(id) { const queued = this.#entry(id, 'QUEUED'); if (queued.reservationBytes > this.#availableBytes()) throw new RangeError('hard quota would be exceeded'); if (queued.writer && this.#leases.has(queued.leaseKey)) throw new Error('exclusive writer lease conflict'); const active = Object.freeze({ ...queued, state: 'ACTIVE', admittedAt: new Date().toISOString() }); this.#entries.set(id, active); this.#reservations.set(id, active); if (active.writer) this.#leases.set(active.leaseKey, id); this.#queuePersist(); return active; }
  attachProcess(id, child) { const active = this.#entry(id, 'ACTIVE'); if (!child || !Number.isSafeInteger(child.pid) || child.pid < 1) throw new TypeError('spawned child pid is required'); const value = Object.freeze({ ...active, ownership: Object.freeze({ pid: child.pid, processGroupId: process.platform === 'win32' ? null : child.pid, platform: process.platform === 'win32' ? 'win32' : 'posix' }) }); this.#entries.set(id, value); this.#reservations.set(id, value); this.#queuePersist(); return value; }
  admit(input) { const queued = this.queue(input); return this.activate(queued.id); }
  release(id, { actualGrowthBytes = 0, exitCode = 0, cache = null, storage = null } = {}) { const growth = byteCount(actualGrowthBytes, 'actualGrowthBytes'); const entry = this.#entry(id, 'ACTIVE'); if (growth > entry.reservationBytes) throw new RangeError('actual growth exceeds reservation'); return this.#terminal(entry, 'RELEASED', { actualGrowthBytes: growth, exitCode, cache, storage }); }
  fail(id, reason, { actualGrowthBytes = 0, exitCode = 125, cache = null, storage = null } = {}) { if (!['unmanaged-output', 'execution-failure', 'reservation-overrun', 'broker-restart', 'activation-failure'].includes(reason)) throw new TypeError('recognized failure reason is required'); return this.#terminal(this.#open(id), 'FAILED', { failureReason: reason, actualGrowthBytes: byteCount(actualGrowthBytes, 'actualGrowthBytes'), exitCode: validExit(exitCode), cache, storage }); }
  cancel(id, reason = 'cancel', { actualGrowthBytes = 0, exitCode = 125, cache = null, storage = null } = {}) { if (!['disconnect', 'cancel', 'broker-restart'].includes(reason)) throw new TypeError('recognized cancellation reason is required'); return this.#terminal(this.#open(id), 'CANCELLED', { cancellationReason: reason, actualGrowthBytes: byteCount(actualGrowthBytes, 'actualGrowthBytes'), exitCode: validExit(exitCode), cache, storage }); }
  poison(id, reason) { return ['disconnect', 'cancel'].includes(reason) ? this.cancel(id, reason) : this.fail(id, reason); }
  gcCandidates() { return [...this.#entries.values()].filter((entry) => ['RELEASED', 'FAILED', 'CANCELLED'].includes(entry.state)).sort((a, b) => (a.lruAt ?? a.terminalAt ?? a.admittedAt ?? a.queuedAt).localeCompare(b.lruAt ?? b.terminalAt ?? b.admittedAt ?? b.queuedAt)).map((entry) => Object.freeze({ id: entry.id, namespace: entry.roots.namespace, leaseKey: entry.leaseKey, lruAt: entry.lruAt ?? entry.terminalAt, roots: entry.roots, reason: entry.failureReason ?? entry.cancellationReason ?? 'released-retention', state: entry.state })); }
  async reconcileUsedBytes() { await this.ready(); const measured = await this.#measureUsedBytes(); this.#reconciliation = measured; if (measured.complete) this.#used = Math.min(this.#capacity, measured.bytes); this.#queuePersist({ id: `reconcile-${randomUUID()}`, state: 'RECONCILED', usedBytes: this.#used, complete: measured.complete, scannedEntries: measured.entries }, true); await this.flush(); return this.capacity; }
  executionEnvironment(id, extra = {}) { const entry = this.#active(id); if (!extra || typeof extra !== 'object' || Object.values(extra).some((value) => typeof value !== 'string')) throw new TypeError('extra environment values must be strings'); const r = entry.roots; const home = join(r.temp, 'home'); return Object.freeze({ ...baseEnvironment(), ...entry.request.environment, ...extra, CARGO_HOME: r.cargoHome, CARGO_TARGET_DIR: r.target, CARGO_BUILD_TARGET_DIR: r.target, CARGO_BUILD_BUILD_DIR: r.build, CARGO_INSTALL_ROOT: join(r.build, 'install'), RUSTUP_HOME: r.rustupHome, SCCACHE_DIR: r.sccache, ...(process.platform === 'win32' ? {} : { SCCACHE_SERVER_UDS: r.sccacheSocket }), npm_config_cache: r.packageCache, npm_config_store_dir: join(r.packageCache, 'pnpm-store'), npm_config_state_dir: join(r.packageCache, 'pnpm-state'), npm_config_node_gyp: r.temp, SWIFTPM_MODULECACHE_OVERRIDE: join(r.packageCache, 'swiftpm-module-cache'), SWIFTPM_PACKAGECACHE_PATH: join(r.packageCache, 'swiftpm-packages'), CMAKE_BUILD_DIR: r.target, HOME: home, XDG_CACHE_HOME: join(r.packageCache, 'xdg-cache'), XDG_CONFIG_HOME: join(home, '.config'), TMPDIR: r.temp, TMP: r.temp, TEMP: r.temp }); }
  #active(id) { return this.#entry(id, 'ACTIVE'); }
  #entry(id, state) { const entry = this.#entries.get(id); if (!entry || entry.state !== state) throw new Error(`${state.toLowerCase()} request not found`); return entry; }
  #open(id) { const entry = this.#entries.get(id); if (!entry || !['QUEUED', 'ACTIVE'].includes(entry.state)) throw new Error('open request not found'); return entry; }
  #terminal(entry, state, details) { this.#finish(entry, state); const growth = details.actualGrowthBytes ?? 0; this.#used = Math.min(this.#capacity, this.#used + growth); const now = new Date().toISOString(); const value = Object.freeze({ ...entry, ...details, state, terminalAt: now, lruAt: now, capacity: this.capacity }); this.#entries.set(entry.id, value); this.#queuePersist(value, true); return value; }
  #finish(entry, state) { this.#reservations.delete(entry.id); if (entry.writer && this.#leases.get(entry.leaseKey) === entry.id) this.#leases.delete(entry.leaseKey); this.#entries.set(entry.id, Object.freeze({ ...entry, state })); }
  #reservedBytes() { return [...this.#reservations.values()].reduce((total, entry) => total + entry.reservationBytes, 0); }
  #availableBytes() { return this.#capacity - this.#used - this.#reservedBytes(); }
  async #load() { try { const saved = JSON.parse(await readFile(this.#stateFile, 'utf8')); this.#used = Number.isSafeInteger(saved.usedBytes) ? Math.min(this.#capacity, byteCount(saved.usedBytes, 'persisted usedBytes')) : 0; const recovered = []; let migrated = false; for (const candidate of saved.entries ?? []) if (candidate?.id) { let entry = candidate; if (entry.state === 'POISONED') { const now = entry.poisonedAt ?? new Date().toISOString(); entry = { ...entry, state: 'FAILED', failureReason: entry.poisonReason ?? 'execution-failure', terminalAt: now, lruAt: entry.lruAt ?? now, legacyPoisoned: true }; migrated = true; } if (['ACTIVE', 'QUEUED'].includes(entry.state)) { let recovery = null; if (entry.state === 'ACTIVE' && entry.ownership) { recovery = await reapPersistedProcessTree(entry.ownership); if (!recovery.reaped) throw new Error(`failed to reap recovered process tree for ${entry.id}`); } const now = new Date().toISOString(); entry = { ...entry, state: 'CANCELLED', cancellationReason: 'broker-restart', terminalAt: now, lruAt: now, recoveredReservationBytes: byteCount(entry.reservationBytes, 'recovered reservationBytes'), recovery, exitCode: 125 }; recovered.push(entry); } if (['RELEASED', 'FAILED', 'CANCELLED'].includes(entry.state)) this.#entries.set(entry.id, Object.freeze(entry)); } this.#reconciliation = await this.#measureUsedBytes(); if (this.#reconciliation.complete) this.#used = Math.min(this.#capacity, this.#reconciliation.bytes); if (migrated || recovered.length) await this.#persist(); for (const entry of recovered) await this.#persist(entry, true); } catch (error) { if (error.code !== 'ENOENT') throw error; this.#reconciliation = await this.#measureUsedBytes(); if (this.#reconciliation.complete) this.#used = Math.min(this.#capacity, this.#reconciliation.bytes); } }
  async #persist(receipt, writeReceipt = false) { const state = { version: 2, usedBytes: this.#used, entries: [...this.#entries.values()] }; await durableWrite(this.#stateFile, `${JSON.stringify(state)}\n`); if (writeReceipt && receipt) await durableWrite(join(this.#receiptsDir, `${receipt.id}.json`), `${JSON.stringify(receipt)}\n`); }
  #queuePersist(receipt = null, writeReceipt = false) { this.#writes = this.#writes.then(() => this.#ready).then(() => this.#persist(receipt, writeReceipt)); }
  async gcApply({ expiredBefore = null, pressure = false } = {}) {
    await this.ready();
    if (this.#gcRun) return this.#gcRun;
    let finish;
    this.#gcRun = new Promise((resolve) => { finish = resolve; });
    try {
      const candidates = this.gcCandidates().filter((entry) => pressure || (expiredBefore && entry.lruAt < expiredBefore)); const deleted = []; let reclaimedBytes = 0; const seenNamespaces = new Set();
      for (const candidate of candidates) {
        if (seenNamespaces.has(candidate.namespace)) continue; seenNamespaces.add(candidate.namespace);
        let namespaceEntries = [...this.#entries.values()].filter((entry) => entry.roots?.namespace === candidate.namespace);
        let live = namespaceEntries.some((entry) => ['ACTIVE', 'QUEUED'].includes(entry.state) || (!pressure && expiredBefore && entry.lruAt >= expiredBefore));
        if (!contained(this.#root.path, candidate.roots.root) || this.#leases.has(candidate.leaseKey) || live) continue;
        const bytes = await directorySize(candidate.roots.root);
        namespaceEntries = [...this.#entries.values()].filter((entry) => entry.roots?.namespace === candidate.namespace);
        live = namespaceEntries.some((entry) => ['ACTIVE', 'QUEUED'].includes(entry.state) || (!pressure && expiredBefore && entry.lruAt >= expiredBefore));
        if (this.#leases.has(candidate.leaseKey) || live) continue;
        await rm(candidate.roots.root, { recursive: true, force: true });
        for (const entry of namespaceEntries) { this.#entries.delete(entry.id); deleted.push(entry.id); }
        reclaimedBytes += bytes;
      }
      const measured = await this.#measureUsedBytes(); this.#reconciliation = measured; if (measured.complete) this.#used = Math.min(this.#capacity, measured.bytes);
      await this.#persist({ id: `gc-${randomUUID()}`, state: 'GC', reclaimedBytes, deleted, usedBytes: this.#used, complete: measured.complete, scannedEntries: measured.entries }, true);
      return { reclaimedBytes, deleted };
    } finally { const done = finish; this.#gcRun = null; done(); }
  }
  async #measureUsedBytes() { return boundedDirectorySize(join(this.#root.path, 'rightkit-build-control'), { maxEntries: 20_000, deadlineMs: 1_000 }); }
}

export class LocalNdjsonBrokerService {
  #broker; #tools = new Map(); #spawn; #statfs; #defaults; #outputPolicy; #bounds; #runtime; #scheduler; #sccache = null; #gcTimer = null; #gcWork = Promise.resolve(); #lastGcAt = 0; #stopping = false; #inflight = new Map();
  constructor({ broker, tools = [], shims = {}, defaults = {}, outputPolicy = {}, bounds = {}, spawn = nodeSpawn, statfs: statfsOperation = statfs, runtimeIdentity = SOURCE_RUNTIME_IDENTITY, scheduler } = {}) { if (!(broker instanceof BuildControlBroker)) throw new TypeError('broker is required'); if (typeof statfsOperation !== 'function') throw new TypeError('statfs operation is required'); this.#broker = broker; this.#spawn = spawn; this.#statfs = statfsOperation; this.#runtime = normalizeRuntimeIdentity(runtimeIdentity); this.#scheduler = scheduler ?? new MachineFifoScheduler({ heartbeatIntervalMs: bounds.heartbeatIntervalMs ?? 5_000 }); this.#defaults = Object.freeze({ reservationBytes: 0, writer: true, shims, ...defaults }); this.#outputPolicy = Object.freeze({ mode: outputPolicy.mode ?? 'enforce', allowedSourceDirs: outputPolicy.allowedSourceDirs ?? [], scanRoot: outputPolicy.scanRoot ?? 'repository' }); this.#bounds = Object.freeze({ heartbeatIntervalMs: duration(bounds.heartbeatIntervalMs ?? 5_000, 'heartbeatIntervalMs'), diskSampleIntervalMs: duration(bounds.diskSampleIntervalMs ?? 5_000, 'diskSampleIntervalMs'), processGraceMs: nonnegativeDuration(bounds.processGraceMs ?? 5_000, 'processGraceMs'), freeFloorBytes: byteCount(bounds.freeFloorBytes ?? 0, 'freeFloorBytes'), quotaWatermarkBytes: bounds.quotaWatermarkBytes === null || bounds.quotaWatermarkBytes === undefined ? null : byteCount(bounds.quotaWatermarkBytes, 'quotaWatermarkBytes'), retentionMs: bounds.retentionMs === null || bounds.retentionMs === undefined ? null : nonnegativeDuration(bounds.retentionMs, 'retentionMs'), gcIntervalMs: duration(bounds.gcIntervalMs ?? 300_000, 'gcIntervalMs') }); for (const tool of tools) this.registerTool(tool); }
  registerTool(tool) { const verified = immutableToolRoot(tool); this.#tools.set(tool.id ?? verified.path, Object.freeze({ id: tool.id ?? verified.path, ...verified })); }
  async handleNdjson(line, write, onDisconnect) {
    if (typeof line !== 'string' || typeof write !== 'function') throw new TypeError('NDJSON line & writer are required');
    let message;
    try { message = JSON.parse(line); } catch { throw new TypeError('invalid NDJSON message'); }
    return this.execute({ ...message, onDisconnect }, (event) => write(`${JSON.stringify(event)}\n`));
  }
  async execute(message, emit = () => {}) {
    await this.#broker.ready();
    if (this.#stopping) { const value = { type: 'RESULT', exitCode: 125, status: 'REJECTED', error: 'RightKit broker is stopping' }; emit(value); return value; }
    let enriched;
    try { enriched = this.#enrich(message, await resolveRequestRoots(message?.request?.workspace, { scanRoot: this.#outputPolicy.scanRoot })); } catch (error) { const value = { type: 'RESULT', exitCode: 125, status: 'REJECTED', error: error.message }; emit(value); return value; }
    emit(Object.freeze({ type: 'HANDSHAKE', runtime: this.#runtime }));
    const tool = this.#tools.get(enriched.toolId);
    let entry; let ticket; let permit; let child = null; let disconnected = false; let termination = null; let holdPermit = false; let resolveDone; const done = new Promise((resolve) => { resolveDone = resolve; });
    const disconnect = () => {
      disconnected = true;
      if (ticket?.cancel()) { try { this.#broker.cancel(entry.id, 'disconnect'); } catch {} return; }
      if (child && !termination) termination = terminateProcessTree(child, { graceMs: this.#bounds.processGraceMs }).catch(() => ({ reaped: false, escalated: true }));
    };
    const unsubscribe = message.onDisconnect?.(disconnect);
    try {
      await rejectWorkspaceCargoConfiguration(enriched.request.workspace);
      if (!tool) throw new Error('unregistered tool'); const file = await readFile(tool.path); if (digest(file) !== tool.sha256) throw new Error('registered tool digest mismatch');
      const sccache = enriched.sccacheToolId ? this.#tools.get(enriched.sccacheToolId) : null;
      if (enriched.sccacheToolId && !sccache) throw new Error('unregistered sccache tool');
      if (sccache && digest(await readFile(sccache.path)) !== sccache.sha256) throw new Error('registered sccache digest mismatch');
      const rustc = await this.#verifiedOptionalTool(enriched.rustcToolId, 'rustc');
      const rustdoc = await this.#verifiedOptionalTool(enriched.rustdocToolId, 'rustdoc');
      if (this.#stopping) throw new Error('RightKit broker is stopping'); entry = await this.#broker.enqueue(enriched); this.#inflight.set(entry.id, Object.freeze({ cancel: disconnect, done })); await this.#broker.flush();
      if (enriched.buildCapable) {
        ticket = this.#scheduler.acquire({ id: entry.id, repositoryIdentity: entry.repositoryIdentity, onHeartbeat: (heartbeat) => emit(Object.freeze({ type: 'HEARTBEAT', id: entry.id, ...heartbeat })) });
        permit = await ticket.permit;
      } else permit = Object.freeze({ release() {} });
      if (disconnected) throw new QueueCancelledError();
      entry = this.#broker.activate(entry.id); await this.#broker.flush(); if (!enriched.buildCapable) emit(Object.freeze({ type: 'HEARTBEAT', id: entry.id, state: 'ACTIVE', position: 0, at: new Date().toISOString() }));
      await mkdirMany(directoryRoots(entry.roots)); const env = this.#broker.executionEnvironment(entry.id, { ...(sccache ? { RUSTC_WRAPPER: sccache.path } : {}), ...(rustc ? { RUSTC: rustc.path } : {}), ...(rustdoc ? { RUSTDOC: rustdoc.path } : {}) });
      if (sccache && !this.#sccache) this.#sccache = new SccacheSupervisor({ tool: sccache });
      const result = await this.#run(entry, tool, env, enriched, emit, (value) => { child = value; if (disconnected && child && !termination) termination = terminateProcessTree(child, { graceMs: this.#bounds.processGraceMs }).catch(() => ({ reaped: false, escalated: true })); });
      if (termination) assertReaped(await termination);
      let receipt;
      if (disconnected) receipt = this.#broker.cancel(entry.id, 'disconnect', result);
      else if (result.reason) receipt = this.#broker.fail(entry.id, result.reason, result);
      else receipt = this.#broker.release(entry.id, result);
      await this.#broker.flush(); this.#scheduleGc();
      const value = Object.freeze({ type: 'RESULT', id: entry.id, exitCode: receipt.exitCode, status: receipt.state, receipt, violations: [] }); emit(value); return value;
    } catch (error) {
      if (error instanceof QueueCancelledError || error.code === 'RK_QUEUE_CANCELLED') {
        let receipt = this.#broker.entry(entry?.id); try { receipt = this.#broker.cancel(entry.id, 'disconnect'); } catch {} await this.#broker.flush(); const value = { type: 'RESULT', id: entry?.id, exitCode: 125, status: 'CANCELLED', receipt }; emit(value); return value;
      }
      if (error.code === 'RK_PROCESS_NOT_REAPED') { holdPermit = true; const value = { type: 'RESULT', id: entry?.id, exitCode: 125, status: 'REAP_PENDING', error: error.message }; emit(value); return value; }
      let receipt = null; if (entry && ['QUEUED', 'ACTIVE'].includes(this.#broker.entry(entry.id)?.state)) { try { receipt = this.#broker.fail(entry.id, 'activation-failure'); await this.#broker.flush(); } catch {} }
      const value = { type: 'RESULT', id: entry?.id, exitCode: 125, status: receipt?.state ?? 'REJECTED', receipt, error: error.message }; emit(value); return value;
    } finally { if (entry) this.#inflight.delete(entry.id); resolveDone(); if (!holdPermit) permit?.release(); if (typeof unsubscribe === 'function') unsubscribe(); }
  }
  async shutdown() { this.#stopping = true; if (this.#gcTimer) { clearTimeout(this.#gcTimer); this.#gcTimer = null; } const inflight = [...this.#inflight.values()]; for (const request of inflight) request.cancel(); await Promise.all(inflight.map(({ done }) => done)); await this.#gcWork; await this.#broker.flush(); }
  #enrich(message, roots) {
    if (!message || message.type !== 'MANAGED_AGENT' || message.protocol !== BUILD_CONTROL_PROTOCOL) throw new TypeError('MANAGED_AGENT v2 payload is required');
    const runtime = normalizeRuntimeIdentity(message.runtime); if (runtime.version !== this.#runtime.version || runtime.clientDigest !== this.#runtime.clientDigest || runtime.brokerDigest !== this.#runtime.brokerDigest) throw new Error('RightKit client/broker runtime digest mismatch');
    const request = normalizeManagedAgentRequest({ ...message.request, workspace: roots.executionCwd });
    validateManagedRustupInvocation(message.shim, request.argv);
    const toolId = this.#defaults.shims[message.shim]; if (!toolId) throw new Error(`no registered tool for shim ${message.shim ?? '<empty>'}`);
    const requestedTarget = compilationTarget(message.shim, request.argv);
    if (requestedTarget && requestedTarget !== this.#defaults.targetTriple) throw new Error(`managed Cargo target must match broker target ${this.#defaults.targetTriple}`);
    // Every admission-control dimension is broker-owned. A client that reaches
    // the socket may choose its command & cwd, never its identity, namespace,
    // lease, reservation, request id, or toolchain.
    const buildCapable = isBuildCapableRust(message.shim, request.argv);
    const enriched = {
      ...this.#defaults, toolId, request, onDisconnect: message.onDisconnect, id: undefined,
      repositoryIdentity: roots.repositoryRoot, scanRoot: roots.scanRoot,
      toolchain: this.#defaults.toolchain, targetTriple: this.#defaults.targetTriple,
      buildProfile: classifyBuildProfile(request.argv, this.#defaults.buildProfile),
      reservationBytes: buildCapable ? this.#defaults.reservationBytes : 0, buildCapable, writer: buildCapable ? this.#defaults.writer : false
    };
    const storageRoots = this.#broker.rootsFor(enriched);
    validateManagedRustEnvironment(message.observedEnvironment ?? {}, {

const cargoConfigurationRedirect = /^\s*(?:\[alias\]|(?:target-dir|build-dir|rustc|rustdoc|rustc-wrapper|rustc-workspace-wrapper)\s*=)/i;
const targetImportLibraryFlags = /^\s*rustflags\s*=\s*\[\s*"-C"\s*,\s*"link-arg=[A-Za-z0-9_.-]+\.lib"\s*\]\s*(?:#.*)?$/i;
function cargoConfigurationRedirects(source) {
  let section = '';
  for (const line of source.split(/\r?\n/)) {
    const header = line.match(/^\s*\[([^\]]+)\]\s*(?:#.*)?$/);
    if (header) section = header[1];
    if (cargoConfigurationRedirect.test(line)) return true;
    if (section.toLowerCase() === 'build' && /^\s*target\s*=/i.test(line)) return true;
    if (/^\s*(?:rustflags|rustdocflags)\s*=/i.test(line) && (!section.toLowerCase().startsWith('target.') || !targetImportLibraryFlags.test(line))) return true;
  }
  return false;
}
/** Cargo reads .cargo/config.toml from the execution cwd & every ancestor, so
 *  inspecting only the cwd leaves an ancestor free to redirect rustc or the
 *  wrapper past the digest-verified toolchain. */
export async function rejectWorkspaceCargoConfiguration(workspace, { maxBytes = 64 * 1024 } = {}) {
  for (const directory of ancestorDirectories(workspace)) for (const name of ['config', 'config.toml']) {
    const candidate = join(directory, '.cargo', name);
    const metadata = await lstat(candidate).catch((error) => error.code === 'ENOENT' ? null : Promise.reject(error));
    if (!metadata) continue;
    if (!metadata.isFile() || metadata.isSymbolicLink() || metadata.size > maxBytes) throw new Error(`managed Cargo configuration is not an admissible bounded file: ${candidate}`);
    if (cargoConfigurationRedirects(await readFile(candidate, 'utf8'))) throw new Error(`managed Cargo configuration cannot redirect tools, outputs, flags, or aliases: ${candidate}`);
  }
}
function ancestorDirectories(directory) {
  if (typeof directory !== 'string' || !isAbsolute(directory)) throw new TypeError('absolute directory is required');
  const values = []; let current = normalize(directory);
  for (;;) { values.push(current); const parent = dirname(current); if (parent === current) return values; current = parent; }
}

/** The three roots the broker previously collapsed into `request.workspace`.
 *  Repository identity is the canonical Git checkout, never the execution cwd,
 *  so two workspaces in one repository share namespace, target root & writer
 *  lease. Resolved from the filesystem so the broker never needs `git` on
 *  PATH, & through realpath so one checkout reached by a symlinked path cannot
 *  mint a second identity. */
export async function resolveRequestRoots(workspace, { scanRoot: scanPolicy = 'repository', operations = { realpath, lstat } } = {}) {
  if (typeof workspace !== 'string' || !isAbsolute(workspace)) throw new TypeError('absolute workspace is required');
  if (!['repository', 'execution'].includes(scanPolicy)) throw new TypeError('scan root policy must be repository or execution');
  if (typeof operations.realpath !== 'function' || typeof operations.lstat !== 'function') throw new TypeError('filesystem operations are required');
  const executionCwd = await operations.realpath(workspace);
  const repositoryRoot = await gitCheckoutRoot(executionCwd, operations.lstat) ?? executionCwd;
  return Object.freeze({ repositoryRoot, executionCwd, scanRoot: scanPolicy === 'repository' ? repositoryRoot : executionCwd });
}
async function gitCheckoutRoot(directory, inspect) {
  for (const candidate of ancestorDirectories(directory)) {
    // Only a missing marker means "not a checkout". Permission & I/O failures
    // must not silently degrade identity back to the execution cwd.
    const marker = await inspect(join(candidate, '.git')).catch((error) => { if (error.code === 'ENOENT') return null; throw error; });
    if (marker && (marker.isDirectory() || marker.isFile())) return candidate;
  }
  return null;
}
function compilationTarget(shim, argv) {
  if (!['cargo', 'rustc', 'rustdoc'].includes(shim)) return null;
  let target = null;
  for (let index = 0; index < argv.length && argv[index] !== '--'; index += 1) {
    let candidate = null;
    if (argv[index] === '--target') {
      candidate = argv[index + 1];
      if (!candidate || candidate === '--') throw new TypeError('--target requires a target triple');
      index += 1;
    } else if (argv[index].startsWith('--target=')) candidate = argv[index].slice('--target='.length);
    if (candidate !== null) {
      if (!candidate || (target && target !== candidate)) throw new TypeError('one non-empty Cargo target is required');
      target = candidate;
    }
  }
  return target;
}
