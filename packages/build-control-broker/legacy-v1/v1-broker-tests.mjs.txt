const talk = (endpoint, message, onEvent = () => {}) => new Promise((resolve, reject) => { const socket = net.createConnection(endpoint); let buffer = ''; const events = []; socket.once('error', reject); socket.on('data', (chunk) => { buffer += chunk; for (;;) { const index = buffer.indexOf('\n'); if (index < 0) break; const event = JSON.parse(buffer.slice(0, index)); events.push(event); onEvent(event); buffer = buffer.slice(index + 1); } }); socket.once('connect', () => socket.write(`${JSON.stringify(message)}\n`)); socket.once('close', () => resolve(events)); });

test('runtime install writes immutable content-addressed manifest, probes isolated state & atomically selects it', async () => { const root = await mkdtemp(path.join(tmpdir(), 'broker-runtime-')); try { const config = { broker: brokerConfig(root, 100), unixSocket: path.join(root, 'broker.sock'), tools: [], shims: {}, defaults: {} }; const result = await installManagedRuntime({ externalRoot: root, sourcePackagesRoot: path.resolve('..'), config, version: 'test-v1', healthTimeoutMs: 4_000 }); assert.match(await readlink(result.current), /^runtime-[a-f0-9]{64}$/); assert.equal(result.contentDigest.length, 64); const manifest = JSON.parse(await readFile(path.join(result.runtime, 'runtime-manifest.json'), 'utf8')); assert.equal(manifest.version, 'test-v1'); assert.equal(manifest.clientDigest.length, 64); assert.equal(manifest.brokerDigest.length, 64); await assert.rejects(readFile(config.broker.stateFile)); } finally { await rm(root, { recursive: true, force: true }); } });
test('runtime activation failure restores prior selection & invokes coherent service rollback', async () => { const root = await mkdtemp(path.join(tmpdir(), 'broker-runtime-')); try { const config = { broker: brokerConfig(root, 100), unixSocket: path.join(root, 'broker.sock'), tools: [], shims: {}, defaults: {} }; await mkdir(path.join(root, 'runtime-prior')); await symlink('runtime-prior', path.join(root, 'runtime-current')); let rolledBack = false; await assert.rejects(installManagedRuntime({ externalRoot: root, sourcePackagesRoot: path.resolve('..'), config, version: 'bad-v1', postSwitchHealth: async () => { throw new Error('service did not become healthy'); }, rollbackSwitch: async () => { rolledBack = true; } }), /rolled back/); assert.equal(rolledBack, true); assert.equal(await readlink(path.join(root, 'runtime-current')), 'runtime-prior'); await assert.rejects(installManagedRuntime({ externalRoot: root, sourcePackagesRoot: path.resolve('..'), config, version: 'bad-v1' }), /already exists/); } finally { await rm(root, { recursive: true, force: true }); } });
